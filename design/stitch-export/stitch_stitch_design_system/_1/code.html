terminal 实际执行的只读 SQL (AST 解析核验) 
SELECT 
  o.region_name,
  COUNT(DISTINCT o.order_id) AS order_cnt,
  SUM(o.pay_amount) / 10000 AS gross_sales_w,
  SUM(IFNULL(r.refund_amount, 0)) / 10000 AS refund_amount_w,
  (SUM(o.pay_amount) - SUM(IFNULL(r.refund_amount, 0))) / 10000 AS net_sales_w,
  ROUND(SUM(IFNULL(r.refund_amount, 0)) / SUM(o.pay_amount) * 100, 2) AS refund_rate_pct
FROM orders o
LEFT JOIN refunds r ON o.order_id = r.order_id
WHERE o.pay_time >= '2026-08-01 00:00:00'
  AND o.pay_time < '2026-09-01 00:00:00'
  AND o.region_name IN ('East_China', 'South_China')
GROUP BY o.region_name;
